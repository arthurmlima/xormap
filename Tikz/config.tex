% Tikz/config.tex
%
% Shared pgfplots configuration for every plot under Tikz/plots/. Load this
% once in your LaTeX preamble (or right before the first \input of a plot),
% then \input{Tikz/plots/<name>.tex} wherever you want that figure.
%
%   \input{Tikz/config.tex}
%   ...
%   \begin{figure}
%     \centering
%     \input{Tikz/plots/gray_entropy.tex}
%     \caption{Cipher entropy vs. K, grayscale corpus.}
%   \end{figure}
%
% Each plot file references its data with a path relative to Tikz/plots/,
% i.e. ../data/<file>.csv, so keep the data/ and plots/ directories together
% wherever Tikz/ is copied.

\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

% Same palette used by the native C++ report generator (src/plot.cpp),
% so these TikZ figures match the sweep_k*.pdf reports.
\definecolor{xormapOrange}{RGB}{235,104,52}
\definecolor{xormapBlue}{RGB}{42,120,214}
\definecolor{xormapGray}{RGB}{158,158,153}
\definecolor{xormapDark}{RGB}{89,89,77}

\pgfplotsset{
  xormap plot/.style={
    width=11cm, height=6.5cm,
    xlabel={K (state bits)},
    grid=both,
    grid style={line width=0.1pt, draw=black!10},
    major grid style={line width=0.2pt, draw=black!25},
    tick label style={font=\footnotesize},
    label style={font=\small},
    title style={font=\small},
    legend style={font=\footnotesize, draw=black!30, fill=white, fill opacity=0.85,
                  text opacity=1},
    legend pos=outer north east,
    every axis plot/.append style={line width=1.2pt},
  },
  % Faint per-image scatter, matching the "individual images" series in the
  % native PDF reports. Contributes its own legend entry — do not mark it
  % forget plot, or the \legend{} list in each plot file will misalign.
  xormap scatter/.style={
    only marks, mark=*, mark size=0.5pt,
    color=xormapGray, opacity=0.35,
  },
  % Bold mean-over-images line.
  xormap mean line/.style={
    mark=*, mark size=1.4pt, thick,
  },
  % Dashed reference line for a theoretical ideal / critical value.
  xormap ideal line/.style={
    dashed, thick, opacity=0.75, mark=none,
  },
}
